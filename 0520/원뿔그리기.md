솔리드 물체 (원뿔 그리기)
 
- 키보드 조작으로 rotation되어야 함 
- glBegin(GL_TRIANGLE_FAN) 사용 
- #define GL_PI 3.1415f 사용 
- glEnable(GL_CULL_FACE)사용 
- 카메라 위치와 방향은 기본 값
- 16개 조각

```c++

#include <gl/glut.h>
#include <stdio.h>
#include <iostream>

#define GL_PI 3.1415f
bool bCull = true;

```c++
// 특수키
GLfloat xRot = 0.0f;
GLfloat yRot = 0.0f;
void SpecialKey(int key, int x, int y) {
	if (key == GLUT_KEY_UP)
		xRot -= 2.0f;
	if (key == GLUT_KEY_DOWN)
		xRot += 2.0f;
	if (key == GLUT_KEY_LEFT)
		yRot -= 2.0f;
	if (key == GLUT_KEY_RIGHT)
		yRot += 2.0f;

	if (xRot > 360.0f)
		xRot -= 360.0f;
	if (xRot < 0.0f)
		xRot += 360.0f;
	if (yRot > 360.0f)
		yRot -= 360.0f;
	if (yRot < 0.0f)
		yRot += 360.0f;

	glutPostRedisplay();
}

// 일반키
GLfloat xTran = 0.0f;
GLfloat yTran = 0.0f;
void keyboard(unsigned char key, int x, int y) {
	if (key == 'a')
		xTran -= 2.0f;
	else if (key == 'd')
		xTran += 2.0f;
	else if (key == 'w')
		yTran += 2.0f;
	else if (key == 'x')
		yTran -= 2.0f;

	glutPostRedisplay();
}

```c++
void RenderScene(void) {

	std::cout << "RenderScene" << std::endl;

	glClear(GL_COLOR_BUFFER_BIT);

	if (bCull)
		glEnable(GL_CULL_FACE);
	else
		glDisable(GL_CULL_FACE);

	glPushMatrix();

	glRotatef(xRot, 1.0f, 0.0f, 0.0f);
	glRotatef(yRot, 0.0f, 1.0f, 0.0f);
	glTranslatef(xTran, yTran, 0.0f);

	GLfloat x, y, angle;
	int iPivot = 1;
	glBegin(GL_TRIANGLE_FAN);
	glVertex3f(0.0f, 0.0f, -50.0f);
	for (angle = 0.0f; angle < (2.0f * GL_PI); angle += (GL_PI / 8.0f)) {
		x = 50.0f * cos(angle);
		y = 50.0f * sin(angle);

		if ((iPivot % 2) == 0)
			glColor3f(0.0f, 1.0f, 0.0f);
		else
			glColor3f(1.0f, 0.0f, 0.0f);

		iPivot++;
		glVertex3f(x, y, 0.0f);
	}
	glEnd();
 
	glBegin(GL_TRIANGLE_FAN);
	glVertex2f(0.0f, 0.0f);
	for (angle = 2.0f * GL_PI; angle >= 0.0f; angle -= (GL_PI / 8.0f)) {
		x = 50.0f * cos(angle);
		y = 50.0f * sin(angle);

		if ((iPivot % 2) == 0)
			glColor3f(0.0f, 1.0f, 0.0f);
		else
			glColor3f(1.0f, 0.0f, 0.0f);

		iPivot++;
		glVertex2f(x, y);
	}
	glEnd();

	glPopMatrix();
	glutSwapBuffers();
}

```c++
void ChangeSize(GLsizei w, GLsizei h) {
	GLfloat aspectRatio;
	GLfloat nRange = 100.0f;
	if (h == 0)
		h = 1;
	glViewport(0, 0, w, h);
	glMatrixMode(GL_PROJECTION);
	glLoadIdentity();
	aspectRatio = (GLfloat)w / (GLfloat)h;
	std::cout << "aspectRatio" << std::endl;
	if (w <= h)
		glOrtho(-nRange, nRange, -nRange / aspectRatio, nRange / aspectRatio, -nRange, nRange);
	else
		glOrtho(-nRange * aspectRatio, nRange * aspectRatio, -nRange, nRange, -nRange, nRange);
	glMatrixMode(GL_MODELVIEW);
	glLoadIdentity();
}

void SetupRC(void) {
	std::cout << "SetupRC" << std::endl;
	glClearColor(1.0f, 1.0f, 1.0f, 1.0f);
	glShadeModel(GL_FLAT);
}

int main(int argc, char** argv) {
	glutInit(&argc, argv);
	glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB);
	glutInitWindowSize(500, 500);
	glutInitWindowPosition(400, 400);
	glutCreateWindow("simple");

	glutDisplayFunc(RenderScene);
	glutReshapeFunc(ChangeSize);
	glutKeyboardFunc(keyboard); // 일반키
	glutSpecialFunc(SpecialKey); // 특수키

	SetupRC();

	glutMainLoop();
}
