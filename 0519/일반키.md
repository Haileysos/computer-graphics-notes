```c++
GLfloat xTran = 0.0f;
GLfloat yTran = 0.0f;

void keyboard(unsigned char key, int x, int y) {
  
	if (key == 'a')
		xTran -= 2.0f;
	else if (key == 'd')
		xTran += 2.0f;
	else if (key == 'w')
		yTran += 2.0f;
	else if (key == 'x')
		yTran -= 2.0f;

	glutPostRedisplay();
}
```
